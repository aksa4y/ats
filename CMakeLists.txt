cmake_minimum_required(VERSION 3.20)
project(ATSExhaust VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(exhaust_model STATIC vendor/modguard/exhaust_model.cpp)
target_include_directories(exhaust_model PUBLIC vendor/modguard)
set_target_properties(exhaust_model PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(particle_backend STATIC src/particle_backend.cpp)
target_include_directories(particle_backend PUBLIC src PRIVATE vendor/spf)
set_target_properties(particle_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows particle backend requires MSVC x64.")
  endif()
  target_link_libraries(particle_backend PRIVATE bcrypt)
endif()
add_library(ATSExhaust SHARED src/plugin.cpp)
target_include_directories(ATSExhaust PRIVATE vendor/spf)
target_link_libraries(ATSExhaust PRIVATE exhaust_model particle_backend)
set_target_properties(ATSExhaust PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(ATSExhaust PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(exhaust_model_test tests/exhaust_model_test.cpp)
  target_link_libraries(exhaust_model_test PRIVATE exhaust_model)
  add_test(NAME exhaust_model COMMAND exhaust_model_test)
  add_executable(plugin_test tests/plugin_test.cpp src/plugin.cpp)
  target_include_directories(plugin_test PRIVATE vendor/spf)
  target_link_libraries(plugin_test PRIVATE exhaust_model particle_backend)
  add_test(NAME plugin_lifecycle COMMAND plugin_test)
  add_executable(particle_test tests/particle_test.cpp)
  target_include_directories(particle_test PRIVATE src)
  add_test(NAME particle_safety COMMAND particle_test)
  foreach(t exhaust_model_test plugin_test particle_test)
    if(MSVC)
      target_compile_options(${t} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${t} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
